  Step-by-step (pure API, no UI):

  1. Prompt list তৈরি করুন (JSON):

  [
    {"prompt": "A cat walking in garden at sunrise", "ref_image": null},
    {"prompt": "Cinematic mountain landscape", "ref_image": "mountain.jpg"},
    {"prompt": "Cyberpunk city street at night", "ref_image": "cyber.jpg"}
  ]

  2. একটা board create করুন (one-time):

  curl -X POST http://127.0.0.1:8101/api/boards \
    -H "Content-Type: application/json" \
    -d '{"name": "Batch 1"}'
  # Returns: {"id": 5}

  3. Board এ একটা Flow project bind করুন (one-time):

  curl -X POST http://127.0.0.1:8101/api/boards/5/project \
    -H "Content-Type: application/json" \
    -d '{"name": "Batch 1"}'
  # Returns: {"flow_project_id": "abc-123-..."}

  4. প্রতিটা prompt এর জন্য এই loop চালান:

  PROJECT_ID="abc-123-..."

  for prompt_data in prompts.json; do
    # (a) Image generate (text → image)
    IMAGE_RESP=$(curl -s -X POST http://127.0.0.1:8101/api/requests \
      -H "Content-Type: application/json" \
      -d "{
        \"type\": \"gen_image\",
        \"params\": {
          \"prompt\": \"$PROMPT\",
          \"project_id\": \"$PROJECT_ID\",
          \"aspect_ratio\": \"IMAGE_ASPECT_RATIO_LANDSCAPE\",
          \"paygate_tier\": \"PAYGATE_TIER_ONE\"
        }
      }")
    IMAGE_REQ_ID=$(echo $IMAGE_RESP | grep -oP '"id":\K\d+')

    # (b) Wait until image done (poll every 10s)
    while true; do
      STATUS=$(curl -s http://127.0.0.1:8101/api/requests/$IMAGE_REQ_ID | grep -oP '"status":"\K[^"]+')
      [ "$STATUS" = "done" ] && break
      [ "$STATUS" = "failed" ] && { echo "Image failed"; continue 2; }
      sleep 10
    done

    # (c) Extract mediaId
    MEDIA_ID=$(curl -s http://127.0.0.1:8101/api/requests/$IMAGE_REQ_ID | grep -oP '"media_ids":\["\K[^"]+')

    # (d) Video generate (image → 8s video)
    VIDEO_RESP=$(curl -s -X POST http://127.0.0.1:8101/api/requests \
      -H "Content-Type: application/json" \
      -d "{
        \"type\": \"gen_video\",
        \"params\": {
          \"prompt\": \"$PROMPT\",
          \"project_id\": \"$PROJECT_ID\",
          \"start_media_id\": \"$MEDIA_ID\",
          \"aspect_ratio\": \"VIDEO_ASPECT_RATIO_LANDSCAPE\",
          \"video_quality\": \"fast\",
          \"paygate_tier\": \"PAYGATE_TIER_ONE\"
        }
      }")
    VIDEO_REQ_ID=$(echo $VIDEO_RESP | grep -oP '"id":\K\d+')

    # (e) Wait until video done (~60-90 seconds)
    while true; do
      STATUS=$(curl -s http://127.0.0.1:8101/api/requests/$VIDEO_REQ_ID | grep -oP '"status":"\K[^"]+')
      [ "$STATUS" = "done" ] && break
      [ "$STATUS" = "failed" ] && { echo "Video failed"; continue 2; }
      sleep 15
    done

    # (f) Download MP4
    VIDEO_MEDIA_ID=$(curl -s http://127.0.0.1:8101/api/requests/$VIDEO_REQ_ID | grep -oP '"media_ids":\["\K[^"]+')
    curl -o "output_${VIDEO_REQ_ID}.mp4" \
      "http://127.0.0.1:8101/media/$VIDEO_MEDIA_ID"

    echo "✓ Saved output_${VIDEO_REQ_ID}.mp4"
  done
